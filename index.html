    const codes = Array.from(document.querySelectorAll('tbody tr')) .map(row => { const el = row.querySelector('.s-table-body-cell-content, td:nth-child(2)'); return el ? el.innerText.trim() : null; }) .filter(text => text && /^[a-zA-Z0-9]{6,10}$/.test(text)); console.log(`Total codes found and copied: ${codes.length}`); console.log(codes.join('\n')); copy(codes.join('\n'));
